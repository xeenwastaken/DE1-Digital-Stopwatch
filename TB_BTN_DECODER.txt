library ieee;
use ieee.std_logic_1164.all;

entity tb_button_decoder is
end tb_button_decoder;

architecture tb of tb_button_decoder is

    component button_decoder
        port (clk      : in std_logic;
              ce       : in std_logic;
              btn_in   : in std_logic;
              tick_out : out std_logic;
              hold_out : out std_logic);
    end component;

    signal clk      : std_logic;
    signal ce       : std_logic;
    signal btn_in   : std_logic;
    signal tick_out : std_logic;
    signal hold_out : std_logic;

    constant TbPeriod : time := 10 ns; 
    signal TbClock : std_logic := '0';
    signal TbSimEnded : std_logic := '0';

begin

    dut : button_decoder
    port map (clk      => clk,
              ce       => ce,
              btn_in   => btn_in,
              tick_out => tick_out,
              hold_out => hold_out);

    
    TbClock <= not TbClock after TbPeriod/2 when TbSimEnded /= '1' else '0';

   
    clk <= TbClock;

    stimuli : process
    begin
        
        ce <= '0';
        btn_in <= '0';

        
        YOURRESETSIGNAL <= '1';
        wait for 100 ns;
        YOURRESETSIGNAL <= '0';
        wait for 100 ns;

        wait for 100 * TbPeriod;

        TbSimEnded <= '1';
        wait;
    end process;

end tb;

configuration cfg_tb_button_decoder of tb_button_decoder is
    for tb
    end for;
end cfg_tb_button_decoder;
